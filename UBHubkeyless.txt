local KEY_FILE = "UBHub_KeyCache.json"

local CORRECT_KEY = "UBHubKeyKaiwakoqowjGsksjebdjejebpqpieolahs"

-- Hàm kiểm tra xem file key đã lưu còn hạn không (24 giờ = 86400 giây)
local function hasValidSavedKey()
	if readfile and isfile and isfile(KEY_FILE) then
		local success, data = pcall(function()
			return game:GetService("HttpService"):JSONDecode(readfile(KEY_FILE))
		end)
		if success and data and data.key == CORRECT_KEY and data.time then
			if os.time() - data.time < 86400 then
				return true
			end
		end
	end
	return false
end

-- Hàm lưu key lại với mốc thời gian hiện tại
local function saveKey()
	if writefile then
		local data = {
			key = CORRECT_KEY,
			time = os.time()
		}
		pcall(function()
			writefile(KEY_FILE, game:GetService("HttpService"):JSONEncode(data))
		end)
	end
end

-- Hàm thực thi load script chính
local function loadMainScript()
	local success, err = pcall(function()
		loadstring(game:HttpGet("https://raw.githubusercontent.com/TeamUBHub/UBLoader/refs/heads/main/Loader.lua"))()
	end)
	
	if not success then
		warn("Error loading main script: " .. tostring(err))
	end
end

-- Nếu đã lưu key hợp lệ trong vòng 24 giờ thì tự động bỏ qua bảng key và load luôn script
if hasValidSavedKey() then
	loadMainScript()
	return
end

-- Hiệu ứng Hover mượt mà
local function applyHover(btn, originalColor, hoverColor)
	btn.MouseEnter:Connect(function()
		game:GetService("TweenService"):Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = hoverColor}):Play()
	end)
	btn.MouseLeave:Connect(function()
		game:GetService("TweenService"):Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = originalColor}):Play()
	end)
end

applyHover(CheckButton, Color3.fromRGB(255, 140, 66), Color3.fromRGB(255, 165, 95))

-- Hàm hiệu ứng lắc nhẹ (Shake Animation) khi nhập sai key
local function shakeWindow()
	local originalPos = MainFrame.Position
	local tweenService = game:GetService("TweenService")
	local info = TweenInfo.new(0.05, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
	
	for i = 1, 3 do
		tweenService:Create(MainFrame, info, {Position = originalPos + UDim2.new(0, -8, 0, 0)}):Play()
		task.wait(0.05)
		tweenService:Create(MainFrame, info, {Position = originalPos + UDim2.new(0, 8, 0, 0)}):Play()
		task.wait(0.05)
	end
	tweenService:Create(MainFrame, info, {Position = originalPos}):Play()
end

-- Xử lý sự kiện nút Get Key
GetKeyButton.MouseButton1Click:Connect(function()
	if setclipboard then
		setclipboard(GET_KEY_LINK)
		ErrorLabel.TextColor3 = Color3.fromRGB(80, 255, 150)
		ErrorLabel.Text = "Link copied to clipboard!"
	else
		ErrorLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		ErrorLabel.Text = "Your executor does not support setclipboard!"
	end
end)

-- Xử lý sự kiện nút Check Key
CheckButton.MouseButton1Click:Connect(function()
	if TextBox.Text == CORRECT_KEY then
		ErrorLabel.TextColor3 = Color3.fromRGB(80, 255, 150)
		ErrorLabel.Text = "Key correct! Loading..."
		
		-- Lưu key vào tệp cấu hình (cache) tính bằng thời gian hiện tại
		saveKey()
		
		task.wait(0.8)
		
		-- Xóa giao diện
		ScreenGui:Destroy()
		
		-- Chạy script chính
		loadMainScript()
	else
		-- Báo lỗi nhập sai, xóa nội dung trong ô nhập và kích hoạt hiệu ứng lắc nhẹ
		ErrorLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		ErrorLabel.Text = "Invalid Key! Please try again."
		TextBox.Text = "" -- Tự động xóa chữ khi nhập sai
		task.spawn(shakeWindow)
	end
end)
